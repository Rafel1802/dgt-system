/**
 * ═══════════════════════════════════════════════════════════════════════════════
 *  DGT System — Google Blogs Sheet Integration
 *  Google Apps Script Web App  (doPost endpoint)
 *
 *  VERSION: 2.0.0
 *  INSPECTED: Actual spreadsheet structure verified against live data.
 *
 *  HOW IT WORKS:
 *    1. Laravel's GoogleBlogsSyncJob sends a POST request here.
 *    2. This script validates the shared secret.
 *    3. Detects duplicate request_id (idempotency) via PropertiesService.
 *    4. Detects duplicate public links inside the target class block.
 *    5. Finds the first row in the class block where Public Link is empty.
 *    6. Writes: Class | [Doc Link — NEVER TOUCHED] | Public Link | Dated | Website
 *    7. Returns JSON.
 *
 *  SHEET STRUCTURE (verified from live data):
 *    Row 1  = Headers: "Class","Doc Link","Public Link","Dated","Website link","",...
 *    Row 2+ = Data rows
 *    Each class occupies a horizontal block of 5 columns + 1 blank separator (6 total).
 *
 *    Block positions (0-indexed columns):
 *      Group 1  → cols  0–4  (A–E)   → Class 3 per spec (but can hold any class value)
 *      Group 2  → cols  6–10 (H–L)   → Class 2
 *      Group 3  → cols 12–16 (M–Q)   → Class 1
 *      Group 4  → cols 18–22 (S–W)   → Class 5
 *      Group 5  → cols 24–28 (Y–AC)  → Class 6
 *      Group 6  → cols 30–34 (AE–AI) → Class 7
 *
 *    Within each group, column offsets are:
 *      +0 = Class label
 *      +1 = Doc Link  ← NEVER WRITTEN BY THIS SCRIPT
 *      +2 = Public Link
 *      +3 = Dated
 *      +4 = Website Link
 * ═══════════════════════════════════════════════════════════════════════════════
 */

/** Target spreadsheet. */

/** Sheet (tab) name. */

/**

 *

 * Value = zero-indexed column offset of the FIRST column (Class) in that block.
 *
 * Each block spans 5 columns: Class | Doc Link | Public Link | Dated | Website Link
 * The Doc Link column (+1) is NEVER written by this script.
 *
 * To add a new class in the future:
 *   1. Extend the spreadsheet horizontally with a new 5-column group + 1 blank separator.
 *   2. Add an entry here: 'newClass': <start column index>.
 *   3. Update GoogleBlogsSheetService::CLASS_BLOCKS in Laravel accordingly.
 *
 * Column index reference (0=A, 1=B, 2=C … 25=Z, 26=AA, 27=AB, 28=AC … 35=AJ):
 */
var CLASS_CONFIG = {

  '4': { blockStart: 0  },  // Class 4 shares same columns as Class 3

  '1': { blockStart: 12 },  // M (blank separator L=11) → M–Q... wait let me recount

  '7': { blockStart: 30 },  // AE–AI

/**
 * NOTE on column offsets — verified from the live CSV:
 * Header row:  "Class","Doc Link","Public Link","Dated","Website link","", "Class","Doc Link",...
 * Positions:     0        1           2           3          4          5    6        7  ...
 *
 * So:
 *   Group 1: starts at col 0  (Class=0, DocLink=1, PublicLink=2, Dated=3, Website=4, Blank=5)
 *   Group 2: starts at col 6  (Class=6, DocLink=7, PublicLink=8, Dated=9, Website=10, Blank=11)
 *   Group 3: starts at col 12 (Class=12, DocLink=13, PublicLink=14, Dated=15, Website=16, Blank=17)
 *   Group 4: starts at col 18 (Class=18, DocLink=19, PublicLink=20, Dated=21, Website=22, Blank=23)
 *   Group 5: starts at col 24 (Class=24, DocLink=25, PublicLink=26, Dated=27, Website=28, Blank=29)
 *   Group 6: starts at col 30 (Class=30, DocLink=31, PublicLink=32, Dated=33, Website=34, Blank=35)
 *
 * Within each group:
 *   blockStart + 0 = Class
 *   blockStart + 1 = Doc Link  ← SKIPPED, writer fills manually
 *   blockStart + 2 = Public Link
 *   blockStart + 3 = Dated
 *   blockStart + 4 = Website Link
 */

/** Row index of the header (0-based). Data starts at HEADER_ROW + 1. */
var HEADER_ROW_INDEX = 0;

/**
 * How many processed request_ids to cache in PropertiesService for idempotency.
 * Oldest entries are removed when the limit is reached to stay within the 500KB quota.
 */
var IDEMPOTENCY_CACHE_SIZE = 2000;

// ── ❷  ENTRY POINT ───────────────────────────────────────────────────────────

/**
 * Handles POST requests from Laravel.
 * Called automatically by Google when the Web App receives a POST.
 */
function doPost(e) {
  var startTime = new Date();

  try {
    // ── Parse body ──────────────────────────────────────────────────────────
    if (!e || !e.postData || !e.postData.contents) {
      return buildResponse({ success: false, message: 'Empty request body.' });
    }

    var payload;
    try {
      payload = JSON.parse(e.postData.contents);
    } catch (parseErr) {
      Logger.log('JSON parse error: ' + parseErr.message);
      return buildResponse({ success: false, message: 'Invalid JSON payload: ' + parseErr.message });
    }

    // ── Validate secret ─────────────────────────────────────────────────────
    var receivedSecret = payload.secret || '';

    // Constant-time comparison to resist timing attacks
    if (!constantTimeEqual(receivedSecret, API_SECRET)) {
      Logger.log('Unauthorized request — secret mismatch.');
      return buildResponse({ success: false, message: 'Unauthorized.' });
    }

    // ── Extract & validate fields ───────────────────────────────────────────
    var classNum   = String(payload['class']  || '').trim();
    var publicLink = String(payload.public_link || '').trim();
    var dated      = String(payload.date       || '').trim();
    var requestId  = String(payload.request_id || '').trim();
    var action     = String(payload.action     || '').trim();

    if (action !== 'delete' && action !== 'export_csv') {
      var validationError = validateFields(classNum, publicLink, dated);
      if (validationError) {
        return buildResponse({ success: false, message: validationError });
      }
    }

    // Read the website name from the payload (provided by Laravel)
    var websiteDomain = payload.website || extractDomain(publicLink);

    // Normalise the public link for comparison (no trailing slash, lowercase)
    var normLink = normaliseUrl(publicLink);

    // ── Concurrency Lock ────────────────────────────────────────────────────
    // Prevents multiple users from finding the same empty row simultaneously
    var lock = LockService.getScriptLock();
    try {
      lock.waitLock(30000); // Wait up to 30 seconds
    } catch (lockErr) {
      Logger.log('Could not obtain lock: ' + lockErr.message);
      return buildResponse({ success: false, message: 'Server is currently busy. Please try again.' });
    }

    try {
      // ── Idempotency check (request_id deduplication) ───────────────────────
      if (requestId && action !== 'delete' && action !== 'export_csv') {
        var already = getIdempotencyResult(requestId);
        if (already) {
          Logger.log('Idempotency hit for request_id=' + requestId);
          return buildResponse(already);
        }
      }

    // ── Open spreadsheet ────────────────────────────────────────────────────
    var ss;
    try {
      ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    } catch (ssErr) {
      Logger.log('Cannot open spreadsheet: ' + ssErr.message);
      return buildResponse({ success: false, message: 'Cannot open spreadsheet. Check permissions.' });
    }

    var sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      return buildResponse({
        success: false,
        message: 'Sheet "' + SHEET_NAME + '" not found in the spreadsheet.'
      });
    }

    // ── Validate class config ───────────────────────────────────────────────
    if (!CLASS_CONFIG.hasOwnProperty(classNum)) {
      var supported = Object.keys(CLASS_CONFIG).sort().join(', ');
      return buildResponse({
        success: false,
        message: 'Unknown class "' + classNum + '". Supported: ' + supported + '.'
      });
    }

    var blockStart = CLASS_CONFIG[classNum].blockStart;

    // Column indices within this block (0-based from spreadsheet left edge)
    var COL_CLASS   = blockStart + 0;
    // COL_DOC      = blockStart + 1  ← NEVER WRITTEN
    var COL_PUBLIC  = blockStart + 2;
    var COL_DATED   = blockStart + 3;
    var COL_WEBSITE = blockStart + 4;

    // ── Read all data in one call (minimise API calls) ──────────────────────
    var lastRow  = sheet.getLastRow();
    var lastCol  = sheet.getLastColumn();

    // Ensure we read enough columns to cover this block
    var minCols = blockStart + 5;
    if (lastCol < minCols) {
      lastCol = minCols;
    }

    var allData;
    if (lastRow <= HEADER_ROW_INDEX) {
      // Sheet is empty or only has a header — just the header row
      allData = [sheet.getRange(1, 1, 1, lastCol).getValues()[0]];
    } else {
      allData = sheet.getRange(1, 1, lastRow, lastCol).getValues();
    }

    var dataStartRow = HEADER_ROW_INDEX + 1; // rows before this are headers

    // ── Handle DELETE action ────────────────────────────────────────────────
    if (action === 'delete') {
      for (var r = dataStartRow; r < allData.length; r++) {
        var cellPublic = String(allData[r][COL_PUBLIC] || '').trim();
        if (cellPublic !== '' && normaliseUrl(cellPublic) === normLink) {
           var delSheetRow = r + 1;
           var delSheetCol = COL_PUBLIC + 1;
           sheet.getRange(delSheetRow, delSheetCol).setValue(''); // Clear only the public link
           
           // Release lock and return success
           lock.releaseLock();
           return buildResponse({ success: true, message: 'Deleted from Google Sheet.' });
        }
      }
      lock.releaseLock();
      return buildResponse({ success: true, message: 'Link not found in Google Sheet.' });
    }

    // ── Handle EXPORT_CSV action ────────────────────────────────────────────
    if (action === 'export_csv') {
      var targetSpreadsheetId = String(payload.spreadsheet_id || '').trim() || SPREADSHEET_ID;
      try {
        var exportSs = SpreadsheetApp.openById(targetSpreadsheetId);
        var exportSheet = exportSs.getSheetByName(SHEET_NAME);
        if (!exportSheet) {
          return buildResponse({ success: false, message: 'Sheet "' + SHEET_NAME + '" not found.' });
        }
        var exportData = exportSheet.getDataRange().getValues();
        var csvString = '';
        for (var i = 0; i < exportData.length; i++) {
          var rowString = exportData[i].map(function(cell) {
             var str = String(cell).replace(/"/g, '""');
             return '"' + str + '"';
          }).join(',');
          csvString += rowString + '\n';
        }
        return buildResponse({ success: true, csv: csvString });
      } catch (err) {
        return buildResponse({ success: false, message: 'Could not open Spreadsheet: ' + err.message });
      }
    }

    // ── Duplicate & Target Row detection ────────────────────────────────────
    var exactMatchEmptyIndex = -1; // Pre-populated row matching Class, Website, Date
    var genericEmptyIndex = -1;    // Fallback row (not reserved for another website)

    for (var r = dataStartRow; r < allData.length; r++) {
      var cellClass   = String(allData[r][COL_CLASS]   || '').trim();
      var cellPublic  = String(allData[r][COL_PUBLIC]  || '').trim();
      var cellDated   = String(allData[r][COL_DATED]   || '').trim();
      var cellWebsite = String(allData[r][COL_WEBSITE] || '').trim();

      if (cellPublic === '') {
        // 1. Is this an EXACT match of pre-populated data?
        if (cellClass === classNum && cellDated === dated && cellWebsite === websiteDomain) {
          if (exactMatchEmptyIndex === -1) exactMatchEmptyIndex = r;
        } 
        // 2. Otherwise, is this a safe blank row? (Not reserved for a different website)
        else if (genericEmptyIndex === -1 && cellWebsite === '') {
          genericEmptyIndex = r;
        }
        // Keep scanning for duplicates in later rows!
        continue;
      }

      var cellNorm = normaliseUrl(cellPublic);
      if (cellNorm === normLink) {
        var dupResult = {
          success:   false,
          duplicate: true,
          message:   'This blog link already exists in Class ' + classNum + ' (row ' + (r + 1) + ').',
          class:     classNum,
          row:       r + 1
        };
        if (requestId) storeIdempotencyResult(requestId, dupResult);
        return buildResponse(dupResult);
      }
    }

    // ── Find write target row ───────────────────────────────────────────────
    var writeRowIndex; // 0-based in allData / sheet

    if (exactMatchEmptyIndex !== -1) {
      // Priority 1: Use the row that was specifically pre-populated for this website and date
      writeRowIndex = exactMatchEmptyIndex;
    } else if (genericEmptyIndex !== -1) {
      // Priority 2: Use the first unreserved blank row in this block
      writeRowIndex = genericEmptyIndex;
    } else {
      // Priority 3: No empty slot found — append after last row
      writeRowIndex = allData.length;
    }

    var writeSheetRow = writeRowIndex + 1; // Convert to 1-based Sheets row number

    // ── Write data ──────────────────────────────────────────────────────────
    // We write each cell individually so we only touch the cells we own.
    // Doc Link (COL_DOC = blockStart + 1) is intentionally NEVER written.

    sheet.getRange(writeSheetRow, COL_CLASS   + 1).setValue(classNum);
    sheet.getRange(writeSheetRow, COL_PUBLIC  + 1).setValue(publicLink);
    sheet.getRange(writeSheetRow, COL_DATED   + 1).setValue(dated);
    sheet.getRange(writeSheetRow, COL_WEBSITE + 1).setValue(websiteDomain);

    SpreadsheetApp.flush();

    var elapsed = new Date() - startTime;
    Logger.log('Blog added: class=' + classNum + ' row=' + writeSheetRow + ' url=' + publicLink + ' (' + elapsed + 'ms)');

    var successResult = {
      success: true,
      message: 'Blog added successfully to Class ' + classNum + ' in row ' + writeSheetRow + '.',
      class:   classNum,
      row:     writeSheetRow
    };

      if (requestId) storeIdempotencyResult(requestId, successResult);
      return buildResponse(successResult);

    } finally {
      // Release lock so others can write
      lock.releaseLock();
    }

  } catch (err) {
    Logger.log('Unhandled error in doPost: ' + err.message + '\n' + err.stack);
    return buildResponse({
      success: false,
      message: 'Internal server error. Please try again. Details: ' + err.message
    });
  }
}

// ── ❹  HELPERS ───────────────────────────────────────────────────────────────

/**
 * Validate required payload fields.
 * Returns an error string, or null if everything is valid.
 */
function validateFields(classNum, publicLink, dated) {
  if (!classNum)   return 'Missing required field: class.';
  if (!publicLink) return 'Missing required field: public_link.';
  if (!dated)      return 'Missing required field: date.';

  // Basic URL sanity check
  if (!/^https?:\/\/.+/i.test(publicLink)) {
    return 'public_link must start with http:// or https://.';
  }

  return null;
}

/**
 * Extract the hostname from a URL, stripping "www." prefix.
 * This runs server-side so the browser cannot spoof the domain.
 *
 * Examples:
 *   https://www.wheelloaders.org/article/  →  wheelloaders.org
 *   https://trenchers.net/page             →  trenchers.net
 */
function extractDomain(url) {
  try {
    var match = url.match(/^https?:\/\/([^\/\?#]+)/i);
    if (!match) return '';
    var host = match[1].toLowerCase();
    // Strip leading www.
    host = host.replace(/^www\./, '');
    // Strip port if present
    host = host.replace(/:\d+$/, '');
    return host;
  } catch (err) {
    return '';
  }
}

/**
 * Normalise a URL for duplicate comparison.
 * - Lowercase
 * - Remove trailing slash(es)
 * - Remove URL fragment (#...)
 */
function normaliseUrl(url) {
  return url.trim().toLowerCase().replace(/#+[^#]*$/, '').replace(/\/+$/, '');
}

/**
 * Constant-time string comparison to prevent timing attacks.
 * Returns true only if a === b exactly.
 */
function constantTimeEqual(a, b) {
  if (a.length !== b.length) return false;
  var result = 0;
  for (var i = 0; i < a.length; i++) {
    result |= a.charCodeAt(i) ^ b.charCodeAt(i);
  }
  return result === 0;
}

/**
 * Build a JSON HTTP response.
 * Note: Apps Script Web Apps always return HTTP 200 at the network level.
 * Callers should inspect the JSON body's "success" field.
 */
function buildResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// ── ❺  IDEMPOTENCY via PropertiesService ─────────────────────────────────────

/**
 * PropertiesService key prefix for idempotency records.
 * Each entry: key = 'idem_<request_id>', value = JSON-encoded result.
 */
var IDEM_PREFIX = 'idem_';

/**
 * Retrieve a previously stored idempotency result for this request_id.
 * Returns the parsed result object, or null if not found.
 */
function getIdempotencyResult(requestId) {
  try {
    var props = PropertiesService.getScriptProperties();
    var key   = IDEM_PREFIX + requestId;
    var stored = props.getProperty(key);
    if (!stored) return null;
    return JSON.parse(stored);
  } catch (err) {
    Logger.log('Idempotency read error: ' + err.message);
    return null; // On error, allow the request to proceed
  }
}

/**
 * Store a result for idempotency lookup on future retries.
 * Evicts the oldest entries when the cache is full.
 */
function storeIdempotencyResult(requestId, result) {
  try {
    var props  = PropertiesService.getScriptProperties();
    var key    = IDEM_PREFIX + requestId;
    var allKeys = props.getKeys().filter(function(k) { return k.indexOf(IDEM_PREFIX) === 0; });

    // Evict oldest entries if we are over the cache size limit
    if (allKeys.length >= IDEMPOTENCY_CACHE_SIZE) {
      // Delete a batch of the oldest entries (they are sorted alphabetically by key,
      // which is fine as request_ids contain timestamps in the Laravel implementation)
      var toDelete = allKeys.slice(0, Math.floor(IDEMPOTENCY_CACHE_SIZE / 4));
      toDelete.forEach(function(k) { props.deleteProperty(k); });
    }

    props.setProperty(key, JSON.stringify(result));
  } catch (err) {
    Logger.log('Idempotency store error (non-fatal): ' + err.message);
    // Non-fatal — the write still happened, we just won't deduplicate retries
  }
}
